# Runtime dependencies for sync.py
#
# PDF -> Markdown conversion (always required). As of pymupdf 2.x the
# pymupdf4llm code ships inside pymupdf itself: pymupdf installs the
# importable `pymupdf4llm` package (the separate pymupdf4llm 2.0 distribution
# is an empty metadata-only placeholder, published to dodge a pip upgrade bug,
# so depending on *it* installs nothing importable).
#
# Pinned exactly: conversion output shifts between releases (heading levels,
# page furniture), and change detection diffs the generated Markdown, so an
# unpinned converter rewrites documents UCI never touched. Every document in
# docs/ was converted with this version — bumping it means re-running
# `sync.py --force` so the whole tree stays on one generation.
pymupdf==2.0

# Documentation site build/deploy. Only invoked by `sync.py --docs` /
# `--ghdeploy`, which shell out to the `properdocs` CLI. properdocs ships no
# themes itself; the Read the Docs theme (set in docs/properdocs.yml) is a
# separate package. Capped below 2.0: a major release here could break the
# build the way pymupdf 2.0 broke the sync.
properdocs>=1.6.7,<2
properdocs-theme-readthedocs>=1.6.5,<2
